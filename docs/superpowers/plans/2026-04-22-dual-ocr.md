**Goal:** Agregar Surya OCR (GPU) corriendo en paralelo con Tesseract (CPU) para enriquecer el contexto que Claude recibe al leer facturas.

**Architecture:** Se agregan tres funciones nuevas en `extractor.py` (`_ocr_surya`, `_ocr_dual`, `_combinar_ocr`) y se modifica `process_file()` para reemplazar la llamada a `_ocr_text()` por `_ocr_dual()`. **Tech Stack:** Python 3.11, `surya-ocr==0.17.1`, `pytesseract`, `concurrent.futures.ThreadPoolExecutor`, `pytest`

def _combinar_ocr(tess: str, surya: str) -> str:
    """Combina textos de dos motores OCR. Si ambos tienen contenido, los etiqueta."""
    t = (tess or "").strip()
    s = (surya or "").strip()
    if t and s:
        return f"=== OCR Tesseract ===\n{t}\n\n=== OCR Surya ===\n{s}"
    return t or s
```

def _ocr_surya(image_path: str) -> str:
    """OCR con Surya (GPU). Retorna '' si Surya no está instalado o falla."""
    try:
        from PIL import Image
        det_predictor, rec_predictor = _get_surya_predictors()
        image = Image.open(image_path).convert("RGB")
        results = rec_predictor([image], [["es"]], det_predictor)
        lines = [line.text for line in results[0].text_lines if line.text]
        text = "\n".join(lines).strip()
        logger.info(f"Surya OCR: {len(lines)} líneas extraídas")
        return text
    except ImportError:
        logger.debug("surya-ocr no disponible — se omite")
        return ""
    except Exception as e:
        logger.warning(f"Surya OCR falló: {e}")
        return ""

def _ocr_dual(image_path: str) -> tuple:
    """Corre Tesseract y Surya en paralelo. Retorna (texto_tess, texto_surya)."""
    import concurrent.futures
    tess_result, surya_result = "", ""
    with concurrent.futures.ThreadPoolExecutor(max_workers=2) as ex:
        f_tess  = ex.submit(_ocr_text, image_path)
        f_surya = ex.submit(_ocr_surya, image_path)
        try:
            tess_result = f_tess.result(timeout=30) or ""
        except Exception as e:
            logger.warning(f"Tesseract falló en paralelo: {e}")
        try:
            surya_result = f_surya.result(timeout=60) or ""
        except Exception as e:
            logger.warning(f"Surya falló en paralelo: {e}")
    logger.info(f"OCR dual — Tesseract: {len(tess_result)} chars, Surya: {len(surya_result)} chars")
    return tess_result, surya_result
```
